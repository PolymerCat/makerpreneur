export type ScheduleInput = {
  subjectId: string;
  subjectName: string;
  examDate: string;
  topics: { id: string; name: string; difficulty: number | null; estimatedHours: number }[];
  availableDays: string[];
  blackoutDates: string[];
  existingDoneDays: string[];
  clusterStart: string | null;
  clusterEnd: string | null;
  revisionDays: number;
};

export type ScheduledDay = {
  date: string;
  topicId: string | null;
  topicName: string | null;
  hours: number;
  blockKind: "study" | "review" | "mock";
};

export function schedulePlan(input: ScheduleInput): ScheduledDay[] {
  var {
    examDate,
    topics,
    availableDays,
    blackoutDates,
    existingDoneDays,
    clusterStart,
    revisionDays,
  } = input;

  var blackouts = new Set(blackoutDates);
  var doneSet = new Set(existingDoneDays);
  var examDt = new Date(examDate);

  var revisionStart = new Date(examDt);
  revisionStart.setDate(revisionStart.getDate() - revisionDays);
  var revisionDateStr = revisionStart.toISOString().substring(0, 10);

  var usableDays = availableDays.filter(function(d) {
    return !blackouts.has(d) && d < examDate && !doneSet.has(d);
  });

  var totalHours = topics.reduce(function(sum, t) { return sum + t.estimatedHours; }, 0);
  if (totalHours === 0 || usableDays.length === 0) return [];

  var hoursPerDay = totalHours / usableDays.length;
  var schedule: ScheduledDay[] = [];
  var topicIdx = 0;
  var remainingHours = topics[topicIdx] ? topics[topicIdx].estimatedHours : 0;

  for (var i = 0; i < usableDays.length; i++) {
    var day = usableDays[i];
    var isRevision = day >= revisionDateStr;

    if (isRevision) {
      schedule.push({
        date: day,
        topicId: null,
        topicName: "Review & Mock",
        hours: 2,
        blockKind: day === revisionDateStr ? "mock" : "review",
      });
      continue;
    }

    var allocated = Math.min(hoursPerDay, remainingHours);
    schedule.push({
      date: day,
      topicId: topics[topicIdx] ? topics[topicIdx].id : null,
      topicName: topics[topicIdx] ? topics[topicIdx].name : null,
      hours: Math.round(allocated * 10) / 10,
      blockKind: "study",
    });

    remainingHours -= allocated;
    if (remainingHours <= 0.01) {
      topicIdx++;
      remainingHours = topicIdx < topics.length ? topics[topicIdx].estimatedHours : 0;
    }
  }

  return schedule;
}

export function computeAvailableDays(
  startDate: string,
  endDate: string,
  weekdayHours: number,
  weekendHours: number,
  blackoutDates: string[]
): string[] {
  var start = new Date(startDate);
  var end = new Date(endDate);
  var days: string[] = [];
  var current = new Date(start);

  while (current <= end) {
    var dayOfWeek = current.getDay();
    var isWeekend = dayOfWeek === 0 || dayOfWeek === 6;
    var hours = isWeekend ? weekendHours : weekdayHours;
    if (hours > 0) {
      days.push(current.toISOString().substring(0, 10));
    }
    current.setDate(current.getDate() + 1);
  }

  return days;
}
```

import { llm } from "./gemini";
import { plannerSequencerPrompt } from "./prompts";

export async function aiBuildTopicSequence(
  subjectName: string,
  examDate: string,
  topics: { name: string; difficulty: number | null; estimatedHours: number }[],
  availableDays: string[],
  clusterStart: string | null,
  revisionDays: number
): Promise<{ dayOffset: number; topicName: string; hours: number }[]> {
  var prompt = plannerSequencerPrompt(subjectName, examDate, topics, availableDays, clusterStart, revisionDays);

  return result;
}

import { NextRequest, NextResponse } from "next/server";
import { createServerSupabaseClient } from "@/lib/supabase/server";
import { aiBuildTopicSequence } from "../../_lib/ai/planner";
import { schedulePlan, computeAvailableDays } from "../../_lib/planner-scheduler";
import { db } from "../../_lib/db";

export async function POST(req: NextRequest): Promise<NextResponse> {
  try {
    var body = await req.json();
    var subjectId: string = body.subjectId;
    var subjectName: string = body.subjectName || "";
    var examDate: string = body.examDate;
    var topics: { name: string; difficulty: number | null; estimatedHours: number }[] = body.topics || [];
    var revisionDays: number = body.revisionDays || 2;

    if (!subjectId || !examDate) {
      return NextResponse.json({ error: "subjectId and examDate required" }, { status: 400 });
    }

    var supabase = await createServerSupabaseClient();
    var { data: userData } = await supabase.auth.getUser();
    var userId = userData?.user?.id || "";

    var settings = await db.getStudySettings(userId);
    var weekdayHours = settings?.weekdayHours ?? 3;
    var weekendHours = settings?.weekendHours ?? 5;
    var blackoutDates: string[] = settings?.blackoutDates ?? [];

    var today = new Date().toISOString().substring(0, 10);
    var availableDays = computeAvailableDays(today, examDate, weekdayHours, weekendHours, blackoutDates);

    var clusterStart: string | null = body.clusterStart ?? null;

    var sequence = await aiBuildTopicSequence(subjectName, examDate, topics, availableDays, clusterStart, revisionDays);

    var planDays = schedulePlan({
      subjectId,
      subjectName,
      examDate,
      topics: topics.map(function(t) { return { ...t, estimatedHours: t.estimatedHours || 2 }; }),
      availableDays,
      blackoutDates,
      existingDoneDays: [],
      clusterStart,
      revisionDays,
    });

    return NextResponse.json({ sequence, planDays });
  } catch (err) {
    console.error("planner generate error:", err);
    return NextResponse.json({ error: "Failed to generate plan" }, { status: 500 });
  }
}
```
